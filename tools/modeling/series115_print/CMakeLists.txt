cmake_minimum_required(VERSION 3.25)
project(KachakachaPrintExport LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
find_package(Qt6 REQUIRED COMPONENTS Core)
find_package(OpenCASCADE REQUIRED)
add_executable(native_print_export export_native.cpp)
target_include_directories(native_print_export PRIVATE ${OpenCASCADE_INCLUDE_DIR})
target_link_libraries(native_print_export PRIVATE Qt6::Core TKernel TKMath TKBRep TKTopAlgo TKGeomBase TKMesh TKDESTL TKDESTEP TKBO)
if(MSVC)
  target_compile_options(native_print_export PRIVATE /utf-8)
endif()
